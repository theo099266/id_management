-- ===========================================
-- DATABASE
-- ===========================================
CREATE DATABASE IF NOT EXISTS template_management_system;
USE template_management_system;

-- ===========================================
-- USERS
-- ===========================================

CREATE TABLE Users
(
    UserID INT AUTO_INCREMENT PRIMARY KEY,

    Name VARCHAR(100) NOT NULL,

    UserName VARCHAR(50) NOT NULL UNIQUE,

    NormalizedUsername VARCHAR(50) NOT NULL UNIQUE,

    Email VARCHAR(150),

    NormalizedEmail VARCHAR(150),

    PasswordHash VARCHAR(255) NOT NULL,

    Role VARCHAR(50) NOT NULL,

    Status VARCHAR(20) DEFAULT 'Active',

    Image LONGBLOB,

    SecurityStamp VARCHAR(255),

    ConcurrencyStamp VARCHAR(255),
	LockoutEnd DATETIME(6) NULL,
	LockoutEnabled TINYINT(1) NOT NULL DEFAULT 0,
	AccessFailedCount INT NOT NULL DEFAULT 0,
    EmailConfirmed TINYINT(1) NOT NULL DEFAULT 0,
	PhoneNumber VARCHAR(50) NULL,
	PhoneNumberConfirmed TINYINT(1) NOT NULL DEFAULT 0,
	TwoFactorEnabled TINYINT(1) NOT NULL DEFAULT 0
);
-- ===========================================
-- TEMPLATES
-- ===========================================

CREATE TABLE Templates (

    TemplateID INT AUTO_INCREMENT PRIMARY KEY,

    TemplateName VARCHAR(150) NOT NULL,

    Category VARCHAR(100),

    Description TEXT,

    TemplateFile VARCHAR(255),

    Version INT DEFAULT 1,

    CreatedBy INT NOT NULL,

    CreatedDate DATETIME,

    CONSTRAINT FK_Template_User
        FOREIGN KEY (CreatedBy)
        REFERENCES Users(UserID)
);

-- ===========================================
-- SIGNATORIES
-- ===========================================

CREATE TABLE Signatories (

    SignatoryID INT AUTO_INCREMENT PRIMARY KEY,

    Name VARCHAR(150),

    Position VARCHAR(150),

    SignatureImage VARCHAR(255),

    CreatedBy INT,

    CONSTRAINT FK_Signatory_User
        FOREIGN KEY (CreatedBy)
        REFERENCES Users(UserID)
);

-- ===========================================
-- BRANDING
-- ===========================================

CREATE TABLE Branding (

    BrandID INT AUTO_INCREMENT PRIMARY KEY,

    OfficeName VARCHAR(150),

    Logo VARCHAR(255),

    Header TEXT,

    Footer TEXT,

    Watermark VARCHAR(255),

    PrimaryColor VARCHAR(20),

    SecondaryColor VARCHAR(20),

    CreatedBy INT,

    CONSTRAINT FK_Branding_User
        FOREIGN KEY (CreatedBy)
        REFERENCES Users(UserID)
);

-- ===========================================
-- DOCUMENTS
-- ===========================================

CREATE TABLE Documents (

    DocumentID INT AUTO_INCREMENT PRIMARY KEY,

    TemplateID INT NOT NULL,

    ReferenceNo VARCHAR(100) NOT NULL,

    Recipient VARCHAR(150),

    Subject VARCHAR(255),

    Body TEXT,

    PreparedBy INT,

    ApprovedBy INT,

    CreatedDate DATETIME,

    GeneratedFile VARCHAR(255),

    CONSTRAINT FK_Document_Template
        FOREIGN KEY (TemplateID)
        REFERENCES Templates(TemplateID),

    CONSTRAINT FK_Document_Prepared
        FOREIGN KEY (PreparedBy)
        REFERENCES Signatories(SignatoryID),

    CONSTRAINT FK_Document_Approved
        FOREIGN KEY (ApprovedBy)
        REFERENCES Signatories(SignatoryID)
);
